function Header() {
  const app = useNightfall();
  const { wide, gutter } = useScreen();
  const profileColor = app.profile?.color ?? C.red;

  if (wide) {
    const links: { label: string; active: boolean; onPress: () => void }[] = [
      { label: 'Home', active: app.tab === 'home' && app.filter === 'all', onPress: () => app.setFilter('all') },
      { label: 'TV Shows', active: app.tab === 'home' && app.filter === 'series', onPress: () => app.setFilter('series') },
      { label: 'Movies', active: app.tab === 'home' && app.filter === 'movie', onPress: () => app.setFilter('movie') },
      { label: 'New & Hot', active: app.tab === 'new', onPress: () => app.setTab('new') },
      { label: 'My Nightfall', active: app.tab === 'my', onPress: () => app.setTab('my') },
    ];
    return (
      <Block direction="row" align="center" gap={28} px={gutter} h={68} bg={C.black}>
        <Text c={C.red} fw="900" size={28} lts={1} onPress={() => app.setFilter('all')}>
          NIGHTFALL
        </Text>
        <Block direction="row" gap={22} flex={1} align="center">
          {links.map((link) => (
            <Text key={link.label} c={link.active ? C.white : C.soft} fw={link.active ? 'bold' : 'normal'} size={14} onPress={link.onPress}>
              {link.label}
            </Text>
          ))}
        </Block>
        <IconButton icon="search" variant="ghost" iconColor={C.white} accessibilityLabel="Search" onPress={() => app.push({ name: 'search' })} />
        <IconButton icon="bell" variant="ghost" iconColor={C.white} accessibilityLabel="Notifications" onPress={() => app.push({ name: 'notifications' })} />
        <Block gap={0} onPress={() => app.setTab('my')} accessibilityRole="button" accessibilityLabel="My Nightfall">
          <ProfileTile color={profileColor} size={32} />
        </Block>
      </Block>
    );
  }

  const heading = app.tab === 'home' ? `For ${app.profile?.name ?? 'You'}` : app.tab === 'new' ? 'New & Hot' : 'My Nightfall';
  return (
    <Block direction="row" align="center" gap={4} px={gutter} h={56} bg={C.black}>
      <Text c={C.white} fw="bold" size={22} flex={1} numberOfLines={1}>
        {heading}
      </Text>
      <IconButton icon="download" variant="ghost" iconColor={C.white} accessibilityLabel="Downloads" onPress={() => app.push({ name: 'downloads' })} />
      <IconButton icon="search" variant="ghost" iconColor={C.white} accessibilityLabel="Search" onPress={() => app.push({ name: 'search' })} />
    </Block>
  );
}

function RouteScreen({ route }: { route: Route }) {
  switch (route.name) {
    case 'title':
      return <TitleScreen id={route.id} />;
    case 'player':
      return <PlayerScreen route={route} />;
    case 'search':
      return <SearchScreen />;
    case 'downloads':
      return <DownloadsScreen />;
    case 'notifications':
      return <NotificationsScreen />;
  }
}

function Nightfall() {
  const app = useNightfall();
  const { wide } = useScreen();
  if (!app.profile) return <ProfilePicker />;

  return (
    <Block flex={1} gap={0} position="relative">
      {/* Tab screens stay mounted under pushed pages so Back returns to the same scroll position. */}
      <Block flex={1} gap={0}>
        <Header />
        <Block flex={1} gap={0} w="100%" maw={wide ? 1400 : undefined} alignSelf="center">
          {app.tab === 'home' && <HomeScreen />}
          {app.tab === 'new' && <NewHotScreen />}
          {app.tab === 'my' && <MyScreen />}
        </Block>
        {!wide && (
          <BottomAppBar
            accessibilityLabel="Main"
            activeKey={app.tab}
            onItemPress={(key) => app.setTab(key as Tab)}
            bg="#121212F2"
            withBorder={false}
            indicator={false}
            activeColor={C.white}
            inactiveColor={C.muted}
            items={[
              {
                key: 'home',
                label: 'Home',
                icon: ({ active, color }) => (active ? <Icon name="home" variant="filled" size={24} color={color} /> : <Icon name="home" size={24} color={color} />),
              },
              {
                key: 'new',
                label: 'New & Hot',
                icon: ({ active, color }) => <Icon name="flame" variant={active ? 'filled' : 'outlined'} size={24} color={color} />,
              },
              {
                key: 'my',
                label: 'My Nightfall',
                icon: () => <ProfileTile color={app.profile?.color ?? C.red} size={24} />,
              },
            ]}
          />
        )}
      </Block>
      {app.stack.map((route, index) => (
        <Block key={`${index}:${JSON.stringify(route)}`} position="absolute" inset={0} zIndex={10 + index} bg={C.black} gap={0}>
          <RouteScreen route={route} />
        </Block>
      ))}
    </Block>
  );
}

export function similarTo(title: Title) {
  return released
    .filter((item) => item.id !== title.id)
    .map((item) => ({ item, score: item.genres.filter((genre) => title.genres.includes(genre)).length + (item.kind === title.kind ? 0.5 : 0) }))

    .map(({ item }) => item);
}

import { createContext, useContext, useEffect, useState, type ReactNode } from 'react';
import AsyncStorage from '@react-native-async-storage/async-storage';

import { byId, PROFILES, type Profile, type Title } from './data';

export type Tab = 'home' | 'new' | 'my';
export type Filter = 'all' | 'series' | 'movie';
export type Rating = 'down' | 'up' | 'love';
export type Route =
  | { name: 'title'; id: string }
  | { name: 'player'; id: string; s?: number; e?: number; trailer?: string }
  | { name: 'search' }
  | { name: 'downloads' }
  | { name: 'notifications' };

interface Saved {
  myList: string[];
  progress: Record<string, number>;
  lastEp: Record<string, [number, number]>;
  ratings: Record<string, Rating>;
  reminders: string[];
  downloads: Record<string, number>;
  trailers: string[];
}

const STORAGE = 'plocks-nightfall-example:v2';
const INITIAL: Saved = {
  myList: ['stillwater', 'afterdark', 'tidekeepers'],
  progress: { wildwood: 0.42, 'nightshift:1:3': 0.19, 'dust:1:2': 0.63, 'nightshift:1:1': 1, 'nightshift:1:2': 1, 'dust:1:1': 1 },
  lastEp: { nightshift: [1, 3], dust: [1, 2] },
  ratings: { evergreen: 'love' },
  reminders: ['northern'],
  downloads: { goldenhour: 1 },
  trailers: ['glasshouse'],
};

export const progressKey = (id: string, s?: number, e?: number) => (s && e ? `${id}:${s}:${e}` : id);

function useNightfallState() {
  const [profile, setProfile] = useState<Profile | null>(null);
  const [tab, setTab] = useState<Tab>('home');
  const [filter, setFilter] = useState<Filter>('all');
  const [genre, setGenre] = useState<string | null>(null);
  const [stack, setStack] = useState<Route[]>([]);
  const [saved, setSaved] = useState<Saved>(INITIAL);
  const [ready, setReady] = useState(false);

  useEffect(() => {
    AsyncStorage.getItem(STORAGE)
      .then((raw) => {
        if (raw) setSaved({ ...INITIAL, ...(JSON.parse(raw) as Partial<Saved>) });
      })
      .catch(() => {})
      .finally(() => setReady(true));
  }, []);
  useEffect(() => {
    if (ready) AsyncStorage.setItem(STORAGE, JSON.stringify(saved)).catch(() => {});
  }, [saved, ready]);

  // Simulated downloads: anything below 1 creeps forward until it finishes.
  const downloading = Object.values(saved.downloads).some((value) => value < 1);
  useEffect(() => {
    if (!downloading) return;
    const timer = setInterval(() => {
      setSaved((old) => ({
        ...old,
        downloads: Object.fromEntries(
          Object.entries(old.downloads).map(([id, value]) => [id, value < 1 ? Math.min(1, value + 0.08) : value]),
        ),
      }));
    }, 350);
    return () => clearInterval(timer);
  }, [downloading]);

  const toggle = (list: string[], id: string) => (list.includes(id) ? list.filter((item) => item !== id) : [...list, id]);
  const update = (patch: (old: Saved) => Partial<Saved>) => setSaved((old) => ({ ...old, ...patch(old) }));

  return {
    ...saved,
    profile,
    tab,
    filter,
    genre,
    stack,
    route: stack[stack.length - 1] as Route | undefined,
    selectProfile: (id: string | null) => {
      setProfile(PROFILES.find((item) => item.id === id) ?? null);
      setStack([]);
      setTab('home');
    },
    setTab: (next: Tab) => {
      setStack([]);
      setTab(next);
    },
    setFilter: (next: Filter, nextGenre: string | null = null) => {
      setStack([]);
      setTab('home');
      setFilter(next);
      setGenre(nextGenre);
    },
    push: (next: Route) => setStack((old) => [...old, next]),
    back: () => setStack((old) => old.slice(0, -1)),
    openTitle: (id: string) => setStack((old) => [...old, { name: 'title', id }]),
    play: (id: string, s?: number, e?: number) => {
      const title = byId(id);
      if (title.kind === 'series' && !s) [s, e] = saved.lastEp[id] ?? [1, 1];
      setStack((old) => [...old, { name: 'player', id, s, e }]);
    },
    playTrailer: (id: string, trailer: string) => {
      update((old) => ({ trailers: old.trailers.includes(id) ? old.trailers : [id, ...old.trailers] }));
      setStack((old) => [...old, { name: 'player', id, trailer }]);
    },
    toggleList: (id: string) => update((old) => ({ myList: toggle(old.myList, id) })),
    toggleReminder: (id: string) => update((old) => ({ reminders: toggle(old.reminders, id) })),
    rate: (id: string, value: Rating) =>
      update((old) => {
        const ratings = { ...old.ratings };
        if (ratings[id] === value) delete ratings[id];
        else ratings[id] = value;
        return { ratings };
      }),
    download: (id: string) => update((old) => ({ downloads: { ...old.downloads, [id]: old.downloads[id] ?? 0 } })),
    removeDownload: (id: string) =>
      update((old) => {
        const downloads = { ...old.downloads };
        delete downloads[id];
        return { downloads };
      }),
    setProgress: (key: string, value: number, id?: string, s?: number, e?: number) =>
      setSaved((old) => {
        if (Math.abs((old.progress[key] ?? 0) - value) < 0.01 && value !== 1) return old;
        const lastEp = id && s && e ? { ...old.lastEp, [id]: [s, e] as [number, number] } : old.lastEp;
        return { ...old, lastEp, progress: { ...old.progress, [key]: value } };
      }),
    /** Progress for the title's resume point (last episode for series). */
    resume: (title: Title) => {
      if (title.kind === 'movie') return { value: saved.progress[title.id] ?? 0, label: '' };
      const [s, e] = saved.lastEp[title.id] ?? [1, 1];
      return { value: saved.progress[progressKey(title.id, s, e)] ?? 0, label: `S${s}:E${e}`, s, e };
    },
  };
}

export type Nightfall = ReturnType<typeof useNightfallState>;
const Context = createContext<Nightfall | null>(null);

export function NightfallProvider({ children }: { children: ReactNode }) {
  return <Context.Provider value={useNightfallState()}>{children}</Context.Provider>;
}

export function useNightfall() {
  const value = useContext(Context);
  if (!value) throw new Error('useNightfall must be used inside NightfallProvider');
  return value;
}

/**
 * Even poster grid. Measures its own width, then picks the column count that keeps
 * posters near `min` wide, so gutters are identical and every poster shares one 2:3 size.
 */
export function PosterGrid({ titles, onPress, min, gap = 8 }: { titles: Title[]; onPress: (id: string) => void; min?: number; gap?: number }) {
  const { poster } = useScreen();
  const [width, setWidth] = useState(0);
  const target = min ?? poster;
  const columns = Math.max(3, Math.floor((width + gap) / (target + gap)));
  const itemW = Math.floor((width - gap * (columns - 1)) / columns);
  return (
    <Block direction="row" wrap="wrap" gap={gap} w="100%" onLayout={(event) => setWidth(event.nativeEvent.layout.width)}>
      {width > 0 && titles.map((item) => <Poster key={item.id} title={item} w={itemW} onPress={() => onPress(item.id)} />)}
    </Block>
  );
}
