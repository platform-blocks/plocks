import { useEffect, useMemo, useState, type ReactNode } from 'react';
import { StatusBar, useWindowDimensions } from 'react-native';
import AsyncStorage from '@react-native-async-storage/async-storage';
import { SafeAreaProvider } from 'react-native-safe-area-context';
import { ChartThemeProvider } from '@plocks/charts';
import { Block, Icon, PlocksProvider, SafeArea, ScrollArea, Tabs, Text, useTheme } from '@plocks/ui';
import type { Side } from './components/TradePanel';
import { findAsset, START_CASH, START_POSITIONS, START_WATCHLIST, type Position } from './data';
import { AssetScreen } from './screens/AssetScreen';
import { AccountScreen, HomeScreen, SearchScreen, WatchlistScreen, type Activity } from './screens/TabScreens';
import { C } from './theme';

type Tab = 'home' | 'search' | 'watchlist' | 'account';
const TABS: Tab[] = ['home', 'search', 'watchlist', 'account'];
const TAB_ICONS: Record<Tab, string> = { home: 'home', search: 'search', watchlist: 'bookmark', account: 'avatar' };
const STORAGE = 'plocks-sproutfolio-example:v2';

function SproutfolioScreen() {
  const { width } = useWindowDimensions();
  const wide = width >= 860;
  const [tab, setTab] = useState<Tab>('home');
  const [positions, setPositions] = useState<Record<string, Position>>(START_POSITIONS);
  const [cash, setCash] = useState(START_CASH);
  const [watchlist, setWatchlist] = useState<string[]>(START_WATCHLIST);
  const [activity, setActivity] = useState<Activity[]>([]);
  const [selected, setSelected] = useState<string | null>(null);
  const [query, setQuery] = useState('');
  const [ready, setReady] = useState(false);
  const asset = selected ? findAsset(selected) : undefined;
  const equity = Object.entries(positions).reduce(
    (sum, [ticker, position]) => sum + position.qty * (findAsset(ticker)?.price ?? 0),
    0,
  );

  useEffect(() => {
    AsyncStorage.getItem(STORAGE)
      .then((raw) => {
        if (!raw) return;
        const data = JSON.parse(raw) as {
          positions?: Record<string, Position>;
          cash?: number;
          watchlist?: string[];
          activity?: Activity[];
        };
        if (data.positions && typeof data.positions === 'object') setPositions(data.positions);
        if (typeof data.cash === 'number') setCash(data.cash);
        if (Array.isArray(data.watchlist)) setWatchlist(data.watchlist);
        if (Array.isArray(data.activity)) setActivity(data.activity);
      })
      .catch(() => {})
      .finally(() => setReady(true));
  }, []);
  useEffect(() => {
    if (ready) AsyncStorage.setItem(STORAGE, JSON.stringify({ positions, cash, watchlist, activity })).catch(() => {});
  }, [positions, cash, watchlist, activity, ready]);

  function trade(side: Side, qty: number) {
    if (!asset) return;
    const total = qty * asset.price;
    setPositions((old) => {
      const current = old[asset.ticker] ?? { qty: 0, avgCost: 0 };
      const nextQty = Math.round((current.qty + (side === 'Buy' ? qty : -qty)) * 1e8) / 1e8;
      const avgCost = side === 'Buy' ? (current.qty * current.avgCost + total) / nextQty : current.avgCost;
      const next = { ...old };
      if (nextQty > 0) next[asset.ticker] = { qty: nextQty, avgCost };
      else delete next[asset.ticker];
      return next;
    });
    setCash((old) => old + (side === 'Buy' ? -total : total));
    setActivity((old) => [{ id: `trade-${Date.now()}`, side, ticker: asset.ticker, qty, total }, ...old]);
  }

  const open = (ticker: string) => setSelected(ticker);
  const nav = (
    <Tabs
      navigationOnly
      orientation={wide ? 'vertical' : 'horizontal'}
      variant={wide ? 'chip' : 'line'}
      color={C.darkGreen}
      value={tab}
      onChange={(item) => {
        setTab(item as Tab);
        setSelected(null);
      }}
      items={TABS.map((item) => ({
        key: item,
        label: item[0].toUpperCase() + item.slice(1),
        icon: (
          <Icon
            name={TAB_ICONS[item]}
            color={tab === item && wide ? C.white : tab === item ? C.darkGreen : C.muted}
            size={wide ? 21 : 20}
          />
        ),
        content: null,
      }))}
    />
  );

  return (
    <SafeArea gap={0} flex={1} bg={C.white}>
      <StatusBar barStyle="dark-content" />
      <Block gap={0} flex={1} direction={wide ? 'row' : 'column'} maw={1280} w="100%" alignSelf="center">
        {wide && (
          <Block w={218} p={18} borderRightWidth={1} borderRightColor={C.line} gap={26}>
            <Text c={C.green} fw="bold" size={28}>
              sprout ↗
            </Text>
            {nav}
            <Block gap={0} flex={1} />
            <Text c={C.muted} size={11}>
              Simulated prices and trades
            </Text>
          </Block>
        )}
        <Block gap={0} flex={1}>
          {asset ? (
            <AssetScreen
              key={asset.ticker}
              asset={asset}
              position={positions[asset.ticker]}
              cash={cash}
              portfolioValue={cash + equity}
              watched={watchlist.includes(asset.ticker)}
              wide={wide}
              onBack={() => setSelected(null)}
              onToggleWatch={() =>
                setWatchlist((old) =>
                  old.includes(asset.ticker) ? old.filter((id) => id !== asset.ticker) : [...old, asset.ticker],
                )
              }
              onOpen={open}
              onOpenList={(list) => {
                setQuery(list);
                setTab('search');
                setSelected(null);
              }}
              onTrade={trade}
            />
          ) : (
            <>
              {!wide && (
                <Block
                  gap={0}
                  h={60}
                  px={18}
                  direction="row"
                  justify="space-between"
                  align="center"
                  borderBottomWidth={1}
                  borderBottomColor={C.line}
                >
                  <Text c={C.green} fw="bold" size={25}>
                    sprout ↗
                  </Text>
                  <Text c={C.muted} size={12}>
                    Demo portfolio
                  </Text>
                </Block>
              )}
              <ScrollArea key={tab} flex={1} contentProps={{ align: 'center', px: 20, pb: 35 }}>
                <Block gap={0} w="100%" maw={820}>
                  {tab === 'home' && (
                    <HomeScreen positions={positions} cash={cash} watchlist={watchlist} onOpen={open} wide={wide} />
                  )}
                  {tab === 'search' && <SearchScreen query={query} setQuery={setQuery} onOpen={open} wide={wide} />}
                  {tab === 'watchlist' && <WatchlistScreen watchlist={watchlist} onOpen={open} />}
                  {tab === 'account' && (
                    <AccountScreen positions={positions} cash={cash} activity={activity} onOpen={open} />
                  )}
                </Block>
              </ScrollArea>
            </>
          )}
          {!wide && !asset && (
            <Block gap={0} borderTopWidth={1} borderTopColor={C.line} bg={C.white}>
              {nav}
            </Block>
          )}
        </Block>
      </Block>
    </SafeArea>
  );
}

/** Chart labels inherit the app font and ink; without it SVG text falls back to the browser serif on web. */
function ChartTheme({ children }: { children: ReactNode }) {
  const theme = useTheme();
  const bridge = useMemo(
    () => ({
      textPrimary: C.ink,
      textSecondary: C.muted,
      background: C.white,
      grid: C.line,
      fontFamily: theme.fontFamily,
    }),
    [theme.fontFamily],
  );
  return <ChartThemeProvider hostThemeBridge={bridge}>{children}</ChartThemeProvider>;
}

export default function App() {
  return (
    <SafeAreaProvider>
      <PlocksProvider theme={{ colorScheme: 'light' }}>
        <ChartTheme>
          <SproutfolioScreen />
        </ChartTheme>
      </PlocksProvider>
    </SafeAreaProvider>
  );
}

      <Block
        gap={0}

        h={size}

        bg={C.white}
        borderWidth={1}
        borderColor={C.line}
        align="center"
        justify="center"

      </Block>
    );

  return (
    <Block gap={0} w={size} h={size} radius={size / 2} bg={asset.color} align="center" justify="center">
      <Text c={C.white} fw="bold" size={Math.round(size * 0.5)} lh={Math.round(size * 0.6)}>
        {asset.glyph ?? asset.ticker[0]}
      </Text>
    </Block>
  );
}

import { AreaChart, CandlestickChart, LineChart } from '@plocks/charts';
import { Block } from '@plocks/ui';
import type { History, Range } from '../data';
import { C, money, price } from '../theme';

export type ChartMode = 'line' | 'candles';

/**
 * The stock-page price chart: a scrubbable line by default, OHLC candles on
 * request. Green or red follows the move over the selected range, and the
 * dashed baseline marks where the range opened (yesterday's close on 1D).
 */
export function PriceChart({
  history,
  range,
  mode,
  height = 240,
  label,
}: {
  history: History;
  range: Range;
  mode: ChartMode;
  height?: number;
  label: string;
}) {
  const up = history.end >= history.start;
  const color = up ? C.green : C.red;
  const baseline = { y: history.start, color: C.muted, dashed: true };
  return (
    <Block gap={0} w="100%" h={height}>
      {mode === 'line' ? (
        <LineChart
          key={`line-${range}`}
          data={history.points}
          h={height}
          lineColor={color}
          lineThickness={2.5}
          showPoints={false}
          xAxis={{ show: false }}
          yAxis={{ show: false }}
          grid={{ show: false }}
          referenceLines={[baseline]}
          enableCrosshair
          liveTooltip
          tooltip={{
            show: true,
            formatter: (point) => `${price(point.y)} · ${point.label ?? ''}`,
          }}
          accessibilityLabel={`${label} price over ${range}`}
        />
      ) : (
        <CandlestickChart
          key={`candles-${range}`}
          h={height}
          series={[
            {
              id: 'price',
              name: label,
              data: history.candles,
              colorBull: C.green,
              colorBear: C.red,
              wickColor: C.muted,
            },
          ]}
          xAxis={{ show: false }}
          yAxis={{ show: true, labelFormatter: (value) => money(value, value >= 1000 ? 0 : 2), labelColor: C.muted }}
          grid={{ show: true, color: C.line }}
          enableCrosshair
          liveTooltip
          tooltip={{
            show: true,
            formatter: (candle) =>
              `O ${price(candle.open)}  H ${price(candle.high)}  L ${price(candle.low)}  C ${price(candle.close)}`,
          }}
          accessibilityLabel={`${label} candlestick chart over ${range}`}
        />
      )}
    </Block>
  );
}

/** Portfolio value over time, as a filled area. */
export function PortfolioChart({
  values,
  labels,
  height = 200,
}: {
  values: number[];
  labels: string[];
  height?: number;
}) {
  const up = values[values.length - 1] >= values[0];
  const color = up ? C.green : C.red;
  return (
    <Block gap={0} w="100%" h={height}>
      <AreaChart
        data={values.map((y, x) => ({ x, y, label: labels[x] }))}
        h={height}
        lineColor={color}
        lineThickness={2.5}
        fillColor={color}
        fillOpacity={0.24}
        showPoints={false}
        xAxis={{ show: false }}
        yAxis={{ show: false }}
        grid={{ show: false }}

      />
    </Block>
  );
}

import { useState } from 'react';
import { Block, Button, Input, SegmentedControl, Text } from '@plocks/ui';
import type { Asset, Position } from '../data';
import { C, money, quantity } from '../theme';

export type Side = 'Buy' | 'Sell';

/**
 * Order ticket. Stocks trade in whole shares; coins in fractional amounts. Used
 * inline beside the stock page on wide screens and in a bottom sheet on phones.
 */
export function TradePanel({
  asset,
  position,
  cash,
  initialSide = 'Buy',
  onSubmit,
  onCancel,
}: {
  asset: Asset;
  position?: Position;
  cash: number;
  initialSide?: Side;
  onSubmit: (side: Side, qty: number) => void;
  onCancel?: () => void;
}) {
  const [side, setSide] = useState<Side>(initialSide);
  const [amount, setAmount] = useState(asset.kind === 'crypto' ? '0.01' : '1');
  const crypto = asset.kind === 'crypto';
  const owned = position?.qty ?? 0;
  const qty = Number(amount);
  const total = qty * asset.price;
  const unit = crypto ? asset.ticker : qty === 1 ? 'share' : 'shares';
  const validAmount = Number.isFinite(qty) && qty > 0 && (crypto || Number.isInteger(qty));
  const valid = validAmount && (side === 'Buy' ? total <= cash : qty <= owned + 1e-9);
  return (
    <Block gap={12}>
      <SegmentedControl
        data={[
          { value: 'Buy', label: 'Buy' },
          { value: 'Sell', label: 'Sell', disabled: !owned },
        ]}
        value={side}
        onChange={(value) => setSide(value as Side)}
        color={side === 'Buy' ? C.green : C.red}
        fullWidth
        size="sm"
        accessibilityLabel="Order side"
      />
      <Text c={C.ink} fw="bold" size={18}>
        {side} {asset.ticker}
      </Text>
      <Block gap={0} direction="row" justify="space-between">
        <Text c={C.muted} size={13}>
          Market price
        </Text>
        <Text c={C.ink} fw="semibold" size={13}>
          {money(asset.price)}
        </Text>
      </Block>
      <Input
        variant="outline"
        label={crypto ? `Amount (${asset.ticker})` : 'Shares'}
        value={amount}
        onChangeText={setAmount}
        keyboardType={crypto ? 'decimal-pad' : 'number-pad'}
        placeholder={crypto ? '0.00' : 'Number of shares'}
        mb={0}
        radius={10}
        inputColor={C.ink}
      />
      <Block gap={0} direction="row" justify="space-between">
        <Text c={C.muted} size={13}>
          Estimated {side === 'Buy' ? 'cost' : 'credit'}
        </Text>
        <Text c={C.ink} fw="bold" size={13}>
          {validAmount ? money(total) : '—'}
        </Text>
      </Block>
      <Text c={C.muted} size={12}>
        {side === 'Buy'
          ? `${money(cash)} buying power available`
          : `${quantity(owned)} ${crypto ? asset.ticker : 'shares'} available`}
      </Text>
      <Button
        title={valid ? `Review ${side.toLowerCase()} · ${quantity(qty)} ${unit}` : `${side} ${asset.ticker}`}
        color={side === 'Buy' ? C.green : C.red}
        disabled={!valid}
        fullWidth
        onPress={() => {
          onSubmit(side, qty);
          setAmount(crypto ? '0.01' : '1');
        }}
      />
      {onCancel && <Button title="Cancel" variant="subtle" color={C.ink} fullWidth onPress={onCancel} />}
      <Text c={C.muted} size={11}>
        Simulated order with sample prices. No real trade is placed.
      </Text>
    </Block>
  );
}

function hash(text: string) {
  let h = 2166136261;
  for (let i = 0; i < text.length; i++) {
    h ^= text.charCodeAt(i);
    h = Math.imul(h, 16777619);
  }
  return h >>> 0;
}

export function rng(seed: string) {
  let a = hash(seed);
  return () => {
    a = (a + 0x6d2b79f5) | 0;
    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

function gauss(next: () => number) {
  const u = Math.max(next(), 1e-9);
  return Math.sqrt(-2 * Math.log(u)) * Math.cos(2 * Math.PI * next());
}

const between = (next: () => number, min: number, max: number) => min + (max - min) * next();

function clock(date: Date) {
  const hours = date.getUTCHours();
  const minutes = date.getUTCMinutes();
  return `${hours % 12 || 12}:${String(minutes).padStart(2, '0')} ${hours >= 12 ? 'PM' : 'AM'}`;
}
function timeLabel(range: Range, time: number) {
  // Labels are written in market time (ET), four hours behind the UTC clock.
  const date = new Date(time - 4 * 3600000);
  if (range === '1D') return clock(date);
  if (range === '1W') return `${WEEKDAYS[date.getUTCDay()]} ${clock(date)}`;
  return `${MONTHS[date.getUTCMonth()]} ${date.getUTCDate()}, ${date.getUTCFullYear()}`;
}

const historyCache = new Map<string, History>();

/** A seeded bridge walk that starts at the range's opening price and ends at today's price. */
export function getHistory(asset: Asset, range: Range): History {
  const key = `${asset.ticker}:${range}`;
  const cached = historyCache.get(key);
  if (cached) return cached;
  const spec = SPECS[range];
  const next = rng(key);
  const crypto = asset.kind === 'crypto';
  const vol = crypto ? 2 : 1;
  const end = asset.price;
  const rangeReturn = range === '1D' ? asset.change / 100 : between(next, spec.returns[0] * vol, spec.returns[1] * vol);
  const start = end / (1 + rangeReturn);
  const count = spec.candles * spec.steps;
  const stepSigma = (spec.sigma * vol) / Math.sqrt(count);
  const walk = [0];
  for (let i = 1; i <= count; i++) walk.push(walk[i - 1] + gauss(next) * stepSigma);
  const logStart = Math.log(start);
  const logEnd = Math.log(end);
  const span = crypto && range === '1D' ? DAY : spec.span;
  const values = walk.map((value, i) => {
    const t = i / count;
    return Math.exp(logStart + (logEnd - logStart) * t + value - t * walk[count]);
  });
  const points = values.map((y, i) => ({ x: i, y, label: timeLabel(range, NOW - span * (1 - i / count)) }));
  const baseVolume = averageVolume(asset) / (range === '1D' ? spec.candles : 1);
  const candles: Candle[] = [];
  for (let c = 0; c < spec.candles; c++) {
    const chunk = values.slice(c * spec.steps, (c + 1) * spec.steps + 1);
    const open = chunk[0];
    const close = chunk[chunk.length - 1];
    const wick = 1 + stepSigma * 0.6 * next();
    candles.push({
      x: c,
      open,
      close,
      high: Math.max(...chunk) * wick,
      low: Math.min(...chunk) / wick,
      volume: baseVolume * between(next, 0.55, 1.6) * (range === '1D' && (c < 3 || c > spec.candles - 4) ? 1.8 : 1),
    });
  }
  const history = { points, candles, start, end };
  historyCache.set(key, history);
  return history;
}

export function averageVolume(asset: Asset) {
  const next = rng(`${asset.ticker}:volume`);
  if (asset.kind === 'crypto') return marketCap(asset) * between(next, 0.025, 0.06);
  return (asset.sharesOut ?? 1) * 1e9 * between(next, 0.003, 0.012);
}

  if (asset.kind === 'crypto') return (asset.circulating ?? 0) * asset.price;
  return (asset.sharesOut ?? 1) * 1e9 * asset.price;
}

export function analystRatings(asset: Asset): Ratings {
  const next = rng(`${asset.ticker}:ratings`);
  const total = Math.round(between(next, 24, 58));
  const buyShare = between(next, 0.35, 0.9);
  const buy = Math.round(total * buyShare);
  const sell = Math.round((total - buy) * between(next, 0, 0.35));
  const targetAvg = asset.price * between(next, 1.02, 1.22);
  return {
    buy,
    hold: total - buy - sell,
    sell,
    targetLow: asset.price * between(next, 0.72, 0.92),
    targetAvg,
    targetHigh: targetAvg * between(next, 1.12, 1.3),
  };
}

  estimate: number;
  actual: number;
  revenue: number;
  margin: number;
}
const QUARTERS = ["Q4 '25", "Q1 '26", "Q2 '26", "Q3 '26"];

export function earnings(asset: Asset) {
  const next = rng(`${asset.ticker}:earnings`);
  const annualEps = asset.pe ? asset.price / asset.pe : -asset.price * 0.02;
  const revenueBase = marketCap(asset) / between(next, 5, 11) / 4;
  const baseMargin = asset.pe ? between(next, 12, 34) : between(next, -8, 2);
  const quarters: Quarter[] = QUARTERS.map((label, i) => {
    const estimate = (annualEps / 4) * (0.9 + i * 0.04) * between(next, 0.94, 1.06);
    const beat = next() < 0.75 ? between(next, 0.01, 0.09) : -between(next, 0.01, 0.05);
    return {
      label,
      estimate,
      actual: estimate + Math.abs(estimate) * beat,
      revenue: revenueBase * (0.9 + i * 0.035) * between(next, 0.97, 1.03),
      margin: baseMargin + i * 0.4 + between(next, -1.5, 1.5),
    };
  });
  const last = quarters[quarters.length - 1];
  return {
    quarters,
    next: { label: "Q4 '26", date: 'Jan 28, 2027', estimate: last.estimate * between(next, 1.01, 1.08) },
  };
}

export const money = (value: number, digits = 2) =>
  (value < 0 ? '-$' : '$') +
  Math.abs(value).toLocaleString('en-US', { minimumFractionDigits: digits, maximumFractionDigits: digits });

/** Sub-dollar coins and tokens need more precision than whole-dollar stocks. */
export const price = (value: number) => money(value, value < 1 ? 4 : 2);

export const signed = (value: number) => (value >= 0 ? '+' : '-') + money(Math.abs(value));

export const pct = (value: number) => `${value >= 0 ? '+' : ''}${value.toFixed(2)}%`;

export const compact = (value: number) => {
  const abs = Math.abs(value);
  if (abs >= 1e12) return `${(value / 1e12).toFixed(2)}T`;
  if (abs >= 1e9) return `${(value / 1e9).toFixed(2)}B`;
  if (abs >= 1e6) return `${(value / 1e6).toFixed(2)}M`;
  if (abs >= 1e3) return `${(value / 1e3).toFixed(1)}K`;
  return value.toFixed(0);
};

export const quantity = (value: number) =>
  value.toLocaleString('en-US', { maximumFractionDigits: Number.isInteger(value) ? 0 : 6 });

export const trend = (value: number) => (value >= 0 ? C.darkGreen : C.red);
