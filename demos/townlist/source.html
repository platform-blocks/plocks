/** Purple primary ramp so built-in controls (FileInput, focus rings, Select) match the brand. */

  colorScheme: 'light' as const,

const TABS: { key: Tab; label: string; icon: string }[] = [
  { key: 'browse', label: 'browse', icon: 'search' },
  { key: 'saved', label: 'saved', icon: 'star' },
  { key: 'post', label: 'post', icon: 'plus' },
  { key: 'messages', label: 'messages', icon: 'message' },
];

function RouteView({ route }: { route: Route }) {
  switch (route.name) {
    case 'home':
      return <Home />;
    case 'categories':
      return <Categories section={route.section} />;
    case 'cities':
      return <Cities />;
    case 'results':
      return <Results initial={route.filter} />;
    case 'listing':
      return <ListingView id={route.id} />;
    case 'threads':
      return <Messages />;
    case 'thread':
      return <Messages openId={route.id} />;
    case 'saved':
      return <Saved />;
    case 'post':
      return <Post />;
  }
}

function routeKey(route: Route) {
  switch (route.name) {
    case 'listing':
    case 'thread':
      return `${route.name}:${route.id}`;
    case 'categories':
      return `categories:${route.section ?? ''}`;
    case 'results':
      return `results:${JSON.stringify(route.filter)}`;
    default:
      return route.name;
  }
}

function TabButton({ item, vertical }: { item: (typeof TABS)[number]; vertical?: boolean }) {
  const store = useStore();
  const active = store.tab === item.key;
  const count = item.key === 'messages' ? store.unread : 0;
  return (
    <Block
      onPress={() => store.goTab(item.key)}
      accessibilityRole="tab"
      accessibilityState={{ selected: active }}
      accessibilityLabel={count ? `${item.label}, ${count} unread` : item.label}
      flex={vertical ? undefined : 1}
      direction={vertical ? 'row' : 'column'}
      align="center"
      gap={vertical ? 10 : 2}
      py={vertical ? 9 : 7}
      px={vertical ? 12 : 0}
      radius={vertical ? 6 : 0}
      bg={vertical && active ? C.tint : undefined}
      pressedStyle={{ opacity: 0.6 }}
    >
      <Block gap={0} position="relative">
        <Icon name={item.icon} variant={active ? 'filled' : 'outlined'} color={active ? C.purple : C.muted} size={vertical ? 19 : 22} />
        {count > 0 && !vertical && (
          <Block gap={0} position="absolute" top={-5} right={-10}>
            <Badge size="xs" variant="filled" color={C.red}>
              {count}
            </Badge>
          </Block>
        )}
      </Block>
      <Text c={active ? C.purple : C.muted} fw={active ? 'bold' : 'semibold'} size={vertical ? 14 : 11} flex={vertical ? 1 : undefined}>
        {item.label}
      </Text>
      {count > 0 && vertical && (
        <Badge size="xs" variant="filled" color={C.red}>
          {count}
        </Badge>
      )}
    </Block>
  );
}

function Sidebar() {
  const store = useStore();
  return (
    <Block w={220} gap={4} p={16} borderRightWidth={1} borderRightColor={C.line} bg={C.paler}>

function Shell() {
  const store = useStore();
  const wide = useWide();
  return (
    <SafeArea gap={0} flex={1} bg={wide ? C.pale : C.white}>
      <StatusBar barStyle="dark-content" />
      <Block gap={0} flex={1} direction={wide ? 'row' : 'column'} w="100%" maw={1320} alignSelf="center" bg={C.white}>
        {wide && <Sidebar />}
        <Block gap={0} flex={1} miw={0}>
          <RouteView key={`${store.tab}:${routeKey(store.route)}`} route={store.route} />
        </Block>
        {!wide && (
          <Block gap={0} direction="row" borderTopWidth={1} borderTopColor={C.line} bg={C.white} accessibilityRole="tablist">
            {TABS.map((item) => (
              <TabButton key={item.key} item={item} />
            ))}
          </Block>
        )}

export function Thumb({ listing, w, h, radius = 4 }: { listing: Listing; w: number | string; h: number; radius?: number }) {
  const first = listing.photos[0];
  if (first) return <Image src={photoSource(first)} w={w as number} h={h} radius={radius} resizeMode="cover" alt="" />;
  return (
    <Block gap={4} w={w as number} h={h} radius={radius} bg={C.pale} align="center" justify="center" borderWidth={1} borderColor={C.line}>
      <Icon name={sectionById(listing.section).icon} color={C.faint} size={Math.min(28, h / 3)} />
      {h > 70 && (
        <Text c={C.faint} size={10}>
          no image
        </Text>
      )}
    </Block>
  );
}

export function PriceTag({ listing, size = 13 }: { listing: Listing; size?: number }) {
  const text = priceText(listing);
  if (!text) return null;
  return (
    <Block gap={0} px={6} py={1} radius={3} bg={listing.price === 0 ? C.greenPale : C.pale} alignSelf="flex-start">
      <Text c={listing.price === 0 ? C.green : C.ink} fw="bold" size={size}>
        {text}
      </Text>
    </Block>
  );
}

export function FavoriteButton({ id, size = 20 }: { id: string; size?: number }) {
  const store = useStore();
  const on = store.favorites.includes(id);
  return (
    <IconButton
      icon="star"
      iconVariant={on ? 'filled' : 'outlined'}
      iconColor={on ? '#D49A00' : C.faint}
      iconSize={size}
      variant="ghost"
      size="sm"
      accessibilityLabel={on ? 'Remove from favorites' : 'Add to favorites'}
      onPress={() => store.toggleFavorite(id)}
    />
  );
}

export const money = (value: number) => (value === 0 ? 'free' : '$' + value.toLocaleString('en-US'));
export const priceText = (listing: Pick<Listing, 'price' | 'per'>) =>
  listing.price === null ? '' : money(listing.price) + (listing.price > 0 && listing.per ? listing.per : '');

export function ageText(at: number, now = Date.now()) {
  const minutes = Math.max(0, Math.round((now - at) / MIN));
  if (minutes < 1) return 'just now';
  if (minutes < 60) return `${minutes}m ago`;
  const hours = Math.round(minutes / 60);
  if (hours < 24) return `${hours}h ago`;
  const days = Math.round(hours / 24);
  return days === 1 ? '1 day ago' : `${days} days ago`;
}

export function dateText(at: number) {
  const date = new Date(at);
  const hours = date.getHours();
  const minutes = String(date.getMinutes()).padStart(2, '0');
  return `${MONTHS[date.getMonth()]} ${date.getDate()}, ${hours % 12 || 12}:${minutes}${hours < 12 ? 'am' : 'pm'}`;
}

export function clockText(at: number, now = Date.now()) {
  const date = new Date(at);
  const sameDay = new Date(now).toDateString() === date.toDateString();
  if (sameDay) {
    const hours = date.getHours();
    return `${hours % 12 || 12}:${String(date.getMinutes()).padStart(2, '0')} ${hours < 12 ? 'am' : 'pm'}`;
  }
  return `${MONTHS[date.getMonth()]} ${date.getDate()}`;
}

function Choice({
  label,
  hint,
  selected,
  onPress,
  right,
}: {
  label: string;
  hint?: string;
  selected: boolean;
  onPress: () => void;
  right?: ReactNode;

      accessibilityRole="radio"

      direction="row"
      align="center"
      gap={12}
      py={10}
      px={12}

function Label({ children, required }: { children: ReactNode; required?: boolean }) {

  );
}

                      <IconButton icon="arrow-right" size="sm" variant="ghost" iconColor={C.body} disabled={index === draft.photos.length - 1} accessibilityLabel={`Move image ${index + 1} later`} onPress={() => move(index, 1)} />
                      <IconButton icon="trash" size="sm" variant="ghost" iconColor={C.red} accessibilityLabel={`Remove image ${index + 1}`} onPress={() => remove(photo.id)} />
                    </Block>
                  </Block>
                ))}
              </Block>
            </>
          )}
        </Block>
      )}

      {step === 5 && listing && (
        <Block gap={12}>
          <Block gap={8} direction="row" align="center" p={10} radius={6} bg={C.amberPale}>
            <Icon name="eye" color={C.amber} size={16} />
            <Text c={C.amber} size={13} flex={1}>
              This is how your posting will look. Use back to edit.

          </Block>
          <Block gap={0} p={wide ? 16 : 10} radius={8} borderWidth={1} borderColor={C.lineStrong} borderStyle="dashed">
            <ListingBody listing={listing} preview />
          </Block>
          <Text c={C.muted} size={12}>
            By publishing you agree to the townlist demo terms. Your posting stays on this device.
          </Text>
        </Block>
      )}
    </Screen>
  );
}

      )}

      <Block gap={8} direction="row" align="center" justify="space-between" wrap mt={10} pb={10} borderBottomWidth={1} borderBottomColor={C.line}>

            has image
          </Chip>
          <Chip size="xs" variant={filter.today ? 'light' : 'outline'} color={C.purple} onPress={() => set({ today: !filter.today })} startSection={filter.today ? <Icon name="check" size={12} color={C.purple} /> : undefined}>

          </Chip>

        <Block gap={8} direction="row" align="center">
          <SegmentedControl
            size="xs"
            data={SORTS}
            value={filter.sort ?? 'newest'}
            onChange={(value) => set({ sort: value as Sort })}
            color={C.purple}
            accessibilityLabel="Sort results"
          />
          <SegmentedControl
            size="xs"
            data={[
              { value: 'list', label: <Icon name="list" size={14} color={view === 'list' ? C.white : C.muted} /> },
              { value: 'gallery', label: <Icon name="grid" size={14} color={view === 'gallery' ? C.white : C.muted} /> },

          <Text c={C.muted}>to</Text>
          <Input variant="outline" size="sm" value={filter.max ?? ''} onChangeText={(max) => set({ max: max.replace(/[^0-9]/g, '') })} placeholder="max $" keyboardType="number-pad" mb={0} flex={1} radius={6} accessibilityLabel="Maximum price" />
          <Text c={C.link} size={12} onPress={() => set({ min: undefined, max: undefined })}>
            reset
          </Text>
        </Block>
      )}

      {results.length === 0 ? (
        <EmptyNote icon="search" title="No posts found">
          Try another search, a wider radius, or fewer filters.
          {'\n'}
          <Text c={C.link} size={13} onPress={() => store.push({ name: 'cities' })}>
            change area or radius
          </Text>
        </EmptyNote>
      ) : view === 'list' ? (
        results.map((item) => <ListingRow key={item.id} listing={item} onPress={() => open(item.id)} />)
      ) : (
        <Block gap={10} direction="row" wrap pt={12}>
          {results.map((item) => (
            <ListingTile key={item.id} listing={item} w={tileW} onPress={() => open(item.id)} />
          ))}
        </Block>
      )}
      {store.hidden.length > 0 && results.length > 0 && (
        <Text c={C.faint} size={11} ta="center" mt={16}>
          {store.hidden.length} hidden post{store.hidden.length === 1 ? '' : 's'} not shown
        </Text>
      )}
    </Screen>
  );
}

/** App state for Townlist: navigation stacks, area, listings, favorites, and messages. */
import { createContext, useCallback, useContext, useEffect, useMemo, useRef, useState, type ReactNode } from 'react';
import AsyncStorage from '@react-native-async-storage/async-storage';
import {
  CITIES,
  HOME_CITY,
  LISTINGS,
  THREADS,
  cityById,
  distance,
  type Listing,
  type SectionId,
  type Thread,
} from './data';

export type Tab = 'browse' | 'saved' | 'post' | 'messages';
export type Sort = 'newest' | 'price-asc' | 'price-desc';
export type Filter = {
  query: string;
  section?: SectionId;
  sub?: string;
  hasImage?: boolean;
  today?: boolean;
  min?: string;
  max?: string;
  sort?: Sort;
};
export type Route =
  | { name: 'home' }
  | { name: 'categories'; section?: SectionId }
  | { name: 'cities' }
  | { name: 'results'; filter: Filter }
  | { name: 'listing'; id: string }
  | { name: 'threads' }
  | { name: 'thread'; id: string }
  | { name: 'saved' }
  | { name: 'post' };
export type SavedSearch = { id: string; filter: Filter; cityId: string; radius: number };

const ROOTS: Record<Tab, Route> = {
  browse: { name: 'home' },
  saved: { name: 'saved' },
  post: { name: 'post' },
  messages: { name: 'threads' },
};
const STORAGE = 'plocks-townlist-example:v2';
const DAY = 24 * 60 * 60_000;

type Persisted = {
  posts: Listing[];
  favorites: string[];
  hidden: string[];
  threads: Thread[];
  cityId: string;
  radius: number;
  searches: SavedSearch[];
};

function useTownlistState() {
  const [tab, setTab] = useState<Tab>('browse');
  const [stacks, setStacks] = useState<Record<Tab, Route[]>>({
    browse: [ROOTS.browse],
    saved: [ROOTS.saved],
    post: [ROOTS.post],
    messages: [ROOTS.messages],
  });
  const [posts, setPosts] = useState<Listing[]>(LISTINGS.filter((item) => item.mine));
  const [favorites, setFavorites] = useState<string[]>(['7803', '7804']);
  const [hidden, setHidden] = useState<string[]>([]);
  const [flagged, setFlagged] = useState<string[]>([]);
  const [threads, setThreads] = useState<Thread[]>(THREADS);
  const [cityId, setCityId] = useState(HOME_CITY);
  const [radius, setRadius] = useState(25);
  const [searches, setSearches] = useState<SavedSearch[]>([]);
  const [ready, setReady] = useState(false);
  const replyTimers = useRef<ReturnType<typeof setTimeout>[]>([]);

  useEffect(() => {
    AsyncStorage.getItem(STORAGE)
      .then((raw) => {
        if (!raw) return;
        const data = JSON.parse(raw) as Partial<Persisted>;
        if (Array.isArray(data.posts)) setPosts(data.posts);
        if (Array.isArray(data.favorites)) setFavorites(data.favorites);
        if (Array.isArray(data.hidden)) setHidden(data.hidden);
        if (Array.isArray(data.threads)) setThreads(data.threads);
        if (Array.isArray(data.searches)) setSearches(data.searches);
        if (typeof data.cityId === 'string' && CITIES.some((city) => city.id === data.cityId)) setCityId(data.cityId);
        if (typeof data.radius === 'number') setRadius(data.radius);
      })
      .catch(() => {})
      .finally(() => setReady(true));
    const timers = replyTimers.current;
    return () => timers.forEach(clearTimeout);
  }, []);
  useEffect(() => {
    if (!ready) return;
    const data: Persisted = { posts, favorites, hidden, threads, cityId, radius, searches };
    AsyncStorage.setItem(STORAGE, JSON.stringify(data)).catch(() => {});
  }, [ready, posts, favorites, hidden, threads, cityId, radius, searches]);

  const listings = useMemo(() => [...posts, ...LISTINGS.filter((item) => !item.mine)], [posts]);
  const listingById = useCallback((id: string) => listings.find((item) => item.id === id), [listings]);
  const city = cityById(cityId);

  /** Listings in the current area (city plus anything within the search radius). */
  const inArea = useCallback(
    (item: Listing) => item.cityId === cityId || distance(city, cityById(item.cityId)) <= radius,
    [city, cityId, radius],
  );

  const search = useCallback(
    (filter: Filter, options: { includeHidden?: boolean } = {}) => {
      const words = filter.query.trim().toLowerCase().split(/\s+/).filter(Boolean);
      const min = Number(filter.min);
      const max = Number(filter.max);
      const now = Date.now();
      const found = listings.filter(
        (item) =>
          inArea(item) &&
          (options.includeHidden || !hidden.includes(item.id)) &&
          (!filter.section || item.section === filter.section) &&
          (!filter.sub || item.sub === filter.sub) &&
          (!filter.hasImage || item.photos.length > 0) &&
          (!filter.today || now - item.postedAt < DAY) &&
          (!filter.min || (item.price ?? 0) >= min) &&
          (!filter.max || (item.price ?? 0) <= max) &&
          words.every((word) => `${item.title} ${item.body} ${item.hood}`.toLowerCase().includes(word)),
      );
      const sort = filter.sort ?? 'newest';
      return found.sort((a, b) =>
        sort === 'newest' ? b.postedAt - a.postedAt : sort === 'price-asc' ? (a.price ?? 0) - (b.price ?? 0) : (b.price ?? 0) - (a.price ?? 0),
      );
    },
    [listings, inArea, hidden],
  );

  const stack = stacks[tab];
  const route = stack[stack.length - 1];
  const push = useCallback(
    (next: Route) => setStacks((old) => ({ ...old, [tab]: [...old[tab], next] })),
    [tab],
  );
  const replace = useCallback(
    (next: Route) => setStacks((old) => ({ ...old, [tab]: [...old[tab].slice(0, -1), next] })),
    [tab],
  );
  const back = useCallback(
    () => setStacks((old) => ({ ...old, [tab]: old[tab].length > 1 ? old[tab].slice(0, -1) : old[tab] })),
    [tab],
  );
  const goTab = useCallback(
    (next: Tab, reset = false) => {
      if (next === tab || reset) setStacks((old) => ({ ...old, [next]: [ROOTS[next]] }));
      setTab(next);
    },
    [tab],
  );
  /** Jump to a tab with a route on top of its root. */
  const openIn = useCallback((next: Tab, target: Route) => {
    setStacks((old) => ({ ...old, [next]: [ROOTS[next], target] }));
    setTab(next);
  }, []);

  const toggle = (setter: typeof setFavorites) => (id: string) =>
    setter((old) => (old.includes(id) ? old.filter((item) => item !== id) : [...old, id]));

  function publish(listing: Listing) {
    setPosts((old) => [listing, ...old]);
  }
  function removePost(id: string) {
    setPosts((old) => old.filter((item) => item.id !== id));
  }

  /** Starts (or continues) a conversation about a listing; returns the thread id. */
  function reply(listing: Listing, text: string) {
    const existing = threads.find((item) => item.listingId === listing.id);
    const id = existing?.id ?? `t-${Date.now()}`;
    const message = { id: `${id}-${Date.now()}`, from: 'me' as const, text, at: Date.now() };
    setThreads((old) =>
      existing
        ? old.map((item) => (item.id === id ? { ...item, messages: [...item.messages, message] } : item))
        : [{ id, listingId: listing.id, name: listing.seller, role: 'buying', unread: 0, messages: [message] }, ...old],
    );
    scheduleAutoReply(id, listing.seller);
    return id;
  }
  function send(threadId: string, text: string) {
    const message = { id: `${threadId}-${Date.now()}`, from: 'me' as const, text, at: Date.now() };
    setThreads((old) => old.map((item) => (item.id === threadId ? { ...item, messages: [...item.messages, message] } : item)));
    const thread = threads.find((item) => item.id === threadId);
    if (thread) scheduleAutoReply(threadId, thread.name);
  }
  function scheduleAutoReply(threadId: string, name: string) {
    const timer = setTimeout(() => {
      setThreads((old) =>
        old.map((item) =>
          item.id === threadId
            ? {
                ...item,
                unread: item.unread + 1,
                messages: [
                  ...item.messages,
                  { id: `${threadId}-r${Date.now()}`, from: 'them', text: `Thanks for the message! This is a demo reply from ${name}.`, at: Date.now() },
                ],
              }
            : item,
        ),
      );
    }, 2200);
    replyTimers.current.push(timer);
  }
  function markRead(threadId: string) {
    setThreads((old) => old.map((item) => (item.id === threadId && item.unread ? { ...item, unread: 0 } : item)));
  }
  function saveSearch(filter: Filter) {
    setSearches((old) => [{ id: `s-${Date.now()}`, filter, cityId, radius }, ...old]);
  }
  function removeSearch(id: string) {
    setSearches((old) => old.filter((item) => item.id !== id));
  }
  function runSearch(saved: SavedSearch) {
    setCityId(saved.cityId);
    setRadius(saved.radius);
    openIn('browse', { name: 'results', filter: saved.filter });
  }

  return {
    tab,
    route,
    canGoBack: stack.length > 1,
    push,
    replace,
    back,
    goTab,
    openIn,
    listings,
    posts,
    listingById,
    search,
    inArea,
    city,
    cityId,
    setCityId,
    radius,
    setRadius,
    favorites,
    toggleFavorite: toggle(setFavorites),
    hidden,
    toggleHidden: toggle(setHidden),
    flagged,
    flag: (id: string) => setFlagged((old) => (old.includes(id) ? old : [...old, id])),
    publish,
    removePost,
    threads,
    reply,
    send,
    markRead,
    unread: threads.reduce((sum, item) => sum + item.unread, 0),
    searches,
    saveSearch,
    removeSearch,
    runSearch,
  };
}

export type Store = ReturnType<typeof useTownlistState>;
const StoreContext = createContext<Store | null>(null);

export function StoreProvider({ children }: { children: ReactNode }) {
  const store = useTownlistState();
  return <StoreContext.Provider value={store}>{children}</StoreContext.Provider>;
}

export function useStore() {
  const store = useContext(StoreContext);
  if (!store) throw new Error('useStore needs StoreProvider');
  return store;
}
